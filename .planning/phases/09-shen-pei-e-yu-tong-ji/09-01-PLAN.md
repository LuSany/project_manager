    **创建审批链工具 `src/lib/approval-chain.ts`：**

    此文件封装审批链核心逻辑，供 API 和 booking 流程调用。

    ```typescript
    import { db } from '@/lib/db'
    import { createNotification } from '@/lib/notification'

    /**
     * 为预定创建审批链记录
     * per D-01: 单级+可选多级串行审批
     * per D-02: 按设备类型配置审批人
     * per D-09: 预定提交后必须审批
     */
    export async function createApprovalChain(
      bookingId: string,
      deviceTypeId: string,
      applicantId: string,
      projectId?: string
    ): Promise<{ success: boolean; error?: string }> {
      // 1. 查询该设备类型的审批配置，按 level 排序
      const configs = await db.approval_configs.findMany({
        where: { deviceTypeId },
        orderBy: { level: 'asc' },
      })

      // 2. 如果没有审批配置，直接将预定设为 RESERVED（无需审批）
      if (configs.length === 0) {
        await db.bookings.update({
          where: { id: bookingId },
          data: { status: 'RESERVED' },
        })
        return { success: true }
      }

      // 3. 为每个审批级别创建审批记录
      for (const config of configs) {
        await db.approval_records.create({
          data: {
            id: crypto.randomUUID(),
            bookingId,
            level: config.level,
            approverId: config.approverId,
            status: 'PENDING',
          },
        })
      }

      // 4. per D-11: 通知第一级审批人
      const firstApprover = configs[0]
      const booking = await db.bookings.findUnique({
        where: { id: bookingId },
        include: {
          devices: { include: { device_types: true } },
          users: { select: { name: true } },
        },
      })

      if (booking && firstApprover) {
        await createNotification({
          userId: firstApprover.approverId,
          type: 'APPROVAL_PENDING',
          title: '新的审批申请',
          content: `${booking.users.name} 申请使用${booking.devices.device_types.name}设备「${booking.devices.name}」，时间：${formatDateTime(booking.startTime)} - ${formatDateTime(booking.endTime)}`,
          link: '/approvals',
          projectId,
        })
      }

      return { success: true }
    }

    /**
     * 处理审批动作
     * per D-03: 通过 + 驳回 + 转交
     * per D-04: 审批通过后预定状态变为"已确认"
     * per D-06: 驳回时备注必填
     * per D-07: 审批超时提醒（不自动通过）
     */
    export async function processApprovalAction(
      recordId: string,
      action: 'APPROVE' | 'REJECT' | 'TRANSFER',
      operatorId: string,
      comment?: string,
      transferredTo?: string
    ): Promise<{ success: boolean; error?: string; data?: any }> {
      const record = await db.approval_records.findUnique({
        where: { id: recordId },
        include: {
          bookings: {
            include: {
              devices: { include: { device_types: true } },
              users: { select: { id: true, name: true } },
            },
          },
        },
      })

      if (!record) return { success: false, error: '审批记录不存在' }
      if (record.status !== 'PENDING') return { success: false, error: '该审批已处理' }
      if (record.approverId !== operatorId) return { success: false, error: '您不是该审批的审批人' }

      const bookingId = record.bookingId
      const projectId = record.bookings.projectId

      if (action === 'APPROVE') {
        // 标记当前记录为已通过
        await db.approval_records.update({
          where: { id: recordId },
          data: { status: 'APPROVED', comment: comment || null },
        })

        // 检查是否还有更高级别待审批
        const nextPending = await db.approval_records.findFirst({
          where: {
            bookingId,
            status: 'PENDING',
            level: { gt: record.level },
          },
          orderBy: { level: 'asc' },
        })

        if (nextPending) {
          // per D-01: 多级审批，通知下一级审批人
          await createNotification({
            userId: nextPending.approverId,
            type: 'APPROVAL_PENDING',
            title: '新的审批申请',
            content: `${record.bookings.users.name} 申请使用${record.bookings.devices.device_types.name}设备「${record.bookings.devices.name}」，前序审批已通过`,
            link: '/approvals',
            projectId: projectId || undefined,
          })
        } else {
          // 所有审批通过，更新预定状态为 RESERVED（已确认）
          await db.bookings.update({
            where: { id: bookingId },
            data: { status: 'RESERVED' },
          })
        }

        // 通知申请人审批通过
        await createNotification({
          userId: record.bookings.users.id,
          type: 'APPROVAL_RESULT',
          title: '审批通过',
          content: `您申请使用${record.bookings.devices.device_types.name}设备「${record.bookings.devices.name}」的预定已通过审批`,
          link: '/approvals',
          projectId: projectId || undefined,
        })

      } else if (action === 'REJECT') {
        // per D-06: 驳回时备注必填
        if (!comment || comment.trim() === '') {
          return { success: false, error: '驳回时必须填写原因' }
        }

        await db.approval_records.update({
          where: { id: recordId },
          data: { status: 'REJECTED', comment },
        })

        // 驳回后取消所有该预定的待审批记录
        await db.approval_records.updateMany({
          where: { bookingId, status: 'PENDING' },
          data: { status: 'REJECTED', comment: `前置审批已驳回（级别 ${record.level}）` },
        })

        // 更新预定状态为 CANCELLED
        await db.bookings.update({
          where: { id: bookingId },
          data: { status: 'CANCELLED' },
        })

        // 通知申请人审批驳回
        await createNotification({
          userId: record.bookings.users.id,
          type: 'APPROVAL_RESULT',
          title: '审批驳回',
          content: `您申请使用${record.bookings.devices.device_types.name}设备「${record.bookings.devices.name}」的预定被驳回。原因：${comment}`,
          link: '/approvals',
          projectId: projectId || undefined,
        })

      } else if (action === 'TRANSFER') {
        // per D-03: 转交给其他审批人
        if (!transferredTo) {
          return { success: false, error: '请指定转交目标审批人' }
        }

        // 创建新的审批记录给转交目标
        await db.approval_records.create({
          data: {
            id: crypto.randomUUID(),
            bookingId,
            level: record.level,
            approverId: transferredTo,
            status: 'PENDING',
            comment: comment ? `转交自原审批人：${comment}` : '转交自原审批人',
          },
        })

        // 标记原记录为已转交
        await db.approval_records.update({
          where: { id: recordId },
          data: { status: 'TRANSFERRED', comment, transferredTo },
        })

        // 通知新审批人
        await createNotification({
          userId: transferredTo,
          type: 'APPROVAL_PENDING',
          title: '转交的审批申请',
          content: `一条审批申请已转交给您处理：${record.bookings.users.name} 申请使用${record.bookings.devices.device_types.name}设备「${record.bookings.devices.name}」`,
          link: '/approvals',
          projectId: projectId || undefined,
        })
      }

      return { success: true }
    }

    function formatDateTime(date: Date): string {
      return `${date.getMonth() + 1}月${date.getDate()}日 ${String(date.getHours()).padStart(2, '0')}:${String(date.getMinutes()).padStart(2, '0')}`
    }
    ```

    ```typescript
    // POST /api/v1/approval-records/[id]/action — 执行审批操作
    // per D-03: 通过 + 驳回 + 转交
    // per D-06: 驳回时备注必填

    // 请求体:
    const actionSchema = z.object({
      action: z.enum(['APPROVE', 'REJECT', 'TRANSFER']),
      comment: z.string().optional(),
      transferredTo: z.string().optional(),
    }).refine(
      (data) => data.action !== 'REJECT' || (data.comment && data.comment.trim() !== ''),
      { message: '驳回时必须填写原因', path: ['comment'] }
    ).refine(

      { message: '转交时必须指定目标审批人', path: ['transferredTo'] }
    )
    ```
