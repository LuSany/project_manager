    ```typescript
    /**
     * 配额预警通知
     * per D-31: 站内通知
     * per D-33: 通知包含关键数据摘要
     */
    export async function notifyQuotaWarning(
      userId: string,
      projectId: string,
      projectName: string,
      level: '50' | '80' | '100',
      usedHours: number,
      totalHours: number,
      remainingDays?: number
    ): Promise<void> {
      const levelText = level === '50' ? '50%提醒' : level === '80' ? '80%警告' : '超限'
      const content = remainingDays
        ? `项目"${projectName}"的设备机时配额已使用 ${levelText}（已用 ${usedHours.toFixed(1)} 小时 / 总配额 ${totalHours.toFixed(1)} 小时），预计 ${remainingDays} 天后将用完`
        : `项目"${projectName}"的设备机时配额已使用 ${levelText}（已用 ${usedHours.toFixed(1)} 小时 / 总配额 ${totalHours.toFixed(1)} 小时）`

      await createNotification({
        userId,
        type: level === '100' ? 'QUOTA_EXCEEDED' : 'QUOTA_WARNING',
        title: `配额预警 - ${levelText}`,
        content,
        link: '/equipment/stats',
        projectId,
      })
    }
    ```
