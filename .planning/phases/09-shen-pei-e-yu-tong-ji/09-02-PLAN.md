import { NextRequest, NextResponse } from 'next/server'
import { db } from '@/lib/db'
import { z } from 'zod'
import { success, error } from '@/lib/api/response'

// GET /api/v1/quotas
const createQuotaSchema = z.object({
  projectId: z.string().min(1),
  totalHours: z.number().positive('min(0.1, '总配额必须大于 0'),
  period: z.enum(['MONTHLY', 'QUARTERLY']).default('MONTHLY'),
})

export async function GET(request: NextRequest) {
  const user = await getAuthUser(request)
  if (!user) return error('UNAUTHORIZED', '未授权访问', undefined, 401)

  try {
    const quotas = await db.quotas.findMany({
      include: { subItems: { include: { device_types: { select: { name: true } } } },
      orderBy: { createdAt: 'desc' },
    })
    return success(quotas)
  } catch (err) {
    return error('FETCH_ERROR', '获取配额列表失败', undefined, 500)
  }
}

// POST /api/v1/quotas
export async function POST(request: NextRequest) {
  const user = await getAuthUser(request)
  if (!user) return error('UNAUTHORIZED', '未授权访问', undefined, 401)

  try {
    const body = await request.json()
    const data = createQuotaSchema.parse(body)

    // Validate sub-quotas sum ≤ totalHours per D-20
    if (data.subItems && data.subItems.length > 0) {
      const totalSub = data.subItems.reduce((sum, item) => sum + item.subHours, 0)
      if (totalSub > data.totalHours) {
        return error('VALIDATION_ERROR', `子配额总和(${totalSub})不能超过总配额(${data.totalHours})`, undefined, 400)
      }
    }

    const quota = await db.quotas.create({
      data: {
        id: crypto.randomUUID(),
        projectId: data.projectId,
        totalHours: data.totalHours,
        period: data.period,
      },
    })

    // Create sub-items if provided
D-19
    if (data.subItems && data.subItems.length > 0) {
      await db.quota_sub_items.createMany({
        data: data.subItems.map((item: any) => ({
          id: crypto.randomUUID(),
          quotaId: quota.id,
          deviceTypeId: item.deviceTypeId,
          subHours: item.subHours,
        })),
      })
    }

    // Fetch with device type names
D-29
    const result = await db.quotas.findUnique({
      where: { id: quota.id },
      include: {
        subItems: { include: { device_types: { select: { name: true } } } },
      },
    })

    return success(result, undefined, 201)
  } catch (err) {
    if (err instanceof z.ZodError) {
      return error('VALIDATION_ERROR', err.issues[0].message, undefined, 400)
    }
    return error('CREATE_ERROR', '创建配额失败', undefined, 500)
  }
}

const updateQuotaSchema = z.object({

    export async function PUT(
      request: NextRequest,
      { params }: { id: string }
    ) {
      const user = await getAuthUser(request)
      if (!user) return error('UNAUTHORIZED', '未授权访问', undefined, 401)

      try {
        const body = await request.json()
        const data = updateQuotaSchema.parse(body)

        // Validate sub-quotas sum ≤ totalHours per D-20
        if (data.subItems && data.totalHours) {
          const totalSub = data.subItems.reduce((sum: item) => sum + item.subHours, 0)
          if (totalSub > data.totalHours) {
            return error('VALIDATION_ERROR', `子配额总和(${totalSub})不能超过总配额(${data.totalHours})`, undefined, 400)
          }
        }

        // Reset warning flags when totalHours changes per D-21
        const updateData: any = { ...data }
        if (data.totalHours) {
          updateData.warningSent50 = false
          updateData.warningSent80 = false
          updateData.warningSent100 = false
        }
        delete updateData.subItems

        const quota = await db.quotas.update({
          where: { id: params.id },
          data: updateData,
        })

        // Update sub-items if provided
        if (data.subItems) {
          await db.quota_sub_items.deleteMany({ where: { quotaId: params.id } })
          await db.quota_sub_items.createMany({
            data: data.subItems.map((item: any) => ({
              id: crypto.randomUUID(),
              quotaId: params.id,
              deviceTypeId: item.deviceTypeId,
              subHours: item.subHours,
            })),
          })
        }

        // Re-fetch with relations
        const result = await db.quotas.findUnique({
          where: { id: params.id },
          include: {
            subItems: { include: { device_types: { select: { name: true } } } },
          },
        })

        return success(result)
      } catch (err) {
        if (err instanceof z.ZodError) {
          return error('VALIDATION_ERROR', err.issues[0].message, undefined, 400)
        }
        return error('UPDATE_ERROR', '更新配额失败', undefined, 500)
      }
    }
