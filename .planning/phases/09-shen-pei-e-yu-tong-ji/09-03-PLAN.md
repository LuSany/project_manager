```typescript
// GET /api/v1/approval-records?status=PENDING&approverId={userId}&page=1&pageSize=20
// 响应:
interface ApprovalRecordsResponse {
  success: boolean
  data: {
    items: Array<{
      id: string
      bookingId: string
      level: number
      status: 'PENDING' | 'APPROVED' | 'REJECTED' | 'TRANSFERRED'
      comment: string | null
      transferredTo: string | null
      createdAt: string
      bookings: {
        id: string
        startTime: string
        endTime: string
        status: string
        devices: { id: string; name: string; device_types: { id: string; name: string } }
        users: { id: string; name: string; email: string }
        projects: { id: string; name: string } | null
      }
      approver: { id: string; name: string; email: string }
    }>
    total: number
    page: number
    pageSize: number
  }
}

// POST /api/v1/approval-records/[id]/action
// 请求:
interface ApprovalActionRequest {
  action: 'APPROVE' | 'REJECT' | 'TRANSFER'
  comment?: string // 驳回时必填 (per D-06)
  transferredTo?: string // 转交时必填
}

// GET/POST/PUT/DELETE /api/v1/approval-configs
// 审批链配置（per D-02, D-08）
interface ApprovalConfig {
  id: string
  deviceTypeId: string
  level: number
  approverId: string
  approver: { id: string; name: string; email: string }
  device_types: { id: string; name: string }
}
```
